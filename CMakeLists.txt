    message(FATAL_ERROR "Unsupported platform: ${CMAKE_SYSTEM_NAME}")

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" processor)
if(processor MATCHES "^(arm64|aarch64)$")
    set(fmi3_arch aarch64)
elseif(processor MATCHES "^(amd64|x86_64)$")
    set(fmi3_arch x86_64)
else()
    message(FATAL_ERROR "Unsupported architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()

    message(FATAL_ERROR "Only x86_64 is supported on ${CMAKE_SYSTEM_NAME}")
endif()

function(add_fmu path identifier recipe)
    set(model "${CMAKE_CURRENT_SOURCE_DIR}/${path}")
    set(source "${model}/sources")
    string(REPLACE "/" "_" target "${path}")
    string(REPLACE "-" "_" target "${target}")
    set(includes "${source}" "${source}/template" "${source}/fmi")
    if(path MATCHES "^fmi2/")
        list(APPEND includes "${fmi2_standard_SOURCE_DIR}/headers")
        set(platform "${fmi2_platform}")
    else()
        list(APPEND includes "${fmi3_standard_SOURCE_DIR}/headers")
        set(platform "${fmi3_arch}-${fmi3_system}")
    endif()

    if(recipe STREQUAL "twentysim")
        file(GLOB units CONFIGURE_DEPENDS "${source}/*.c")
        set(definitions FMI_COSIMULATION NO_FUNCTION_PREFIX)
    elseif(recipe STREQUAL "template")
        file(GLOB units CONFIGURE_DEPENDS "${source}/*.c")
        set(definitions FMI_COSIMULATION)
    elseif(recipe STREQUAL "rollback")
        set(units "${source}/fmu.cpp")
        set(definitions FMI_COSIMULATION)
    elseif(recipe STREQUAL "dymola")
        set(units "${source}/all.c")
        set(definitions FMI_2 FMI_COSIMULATION FMI2_FUNCTION_PREFIX=)
    elseif(recipe STREQUAL "reference")
        set(units "${source}/model.c" "${source}/cosimulation.c" "${source}/fmi3Functions.c")
        set(definitions FMI_VERSION=3 DISABLE_PREFIX)
    elseif(recipe STREQUAL "jni2")
        set(units "${source}/fmu2test1.c")
        set(definitions FMI_VERSION=2 "MODEL_GUID=\"{12345678-9999-9999-9999-000000000000}\"" MODEL_IDENTIFIER=fmi2functiontest FMI_COSIMULATION)
    elseif(recipe STREQUAL "jni3")
        set(units "${source}/fmi3functiontest.c")
        set(definitions FMI_VERSION=3)
    elseif(recipe STREQUAL "overture")
        file(GLOB units CONFIGURE_DEPENDS "${source}/*.c" "${source}/vdmlib/*.c")
        list(APPEND includes "${source}/vdmlib")
        set(definitions CUSTOM_IO PERIODIC_GENERATED FMI_COSIMULATION)
    elseif(recipe STREQUAL "openmodelica")
        file(GLOB units CONFIGURE_DEPENDS "${source}/*.c")
        list(FILTER units EXCLUDE REGEX "_info\\.c$")
        set(runtime "${source}/include")
        set(modelica "${CMAKE_CURRENT_SOURCE_DIR}/build-support/modelica-3.2.2")
        list(APPEND includes "${runtime}" "${runtime}/fmi2" "${runtime}/meta" "${runtime}/gc" "${modelica}" "${modelica}/zlib")
        set(definitions OMC_MINIMAL_RUNTIME=1 OMC_FMI_RUNTIME=1 CMINPACK_NO_DLL=1 DUMMY_FUNCTION_USERTAB)
        if(NOT WIN32)
            list(APPEND definitions HAVE_UNISTD_H)
        endif()
        foreach(part ModelicaStandardTables.c ModelicaMatIO.c ModelicaStrings.c)
            list(APPEND units "${modelica}/${part}")
        endforeach()
        file(GLOB extra CONFIGURE_DEPENDS "${modelica}/zlib/*.c" "${CMAKE_CURRENT_SOURCE_DIR}/build-support/openmodelica-d19f816/*.c" "${runtime}/external_solvers/*.c")
        list(APPEND units ${extra})
        foreach(part gc/memory_pool.c math-support/pivot.c meta/meta_modelica_catch.c simulation/simulation_info_json.c simulation/options.c simulation/simulation_omc_assert.c simulation/solver/initialization/initialization.c)
            list(APPEND units "${runtime}/${part}")
        endforeach()
        foreach(name base_array boolean_array omc_error division generic_array index_spec integer_array list modelica_string real_array ringbuffer string_array utility varinfo ModelicaUtilities omc_msvc simulation_options cJSON rational modelica_string_lit omc_init omc_mmap)
            list(APPEND units "${runtime}/util/${name}.c")
        endforeach()
        foreach(name delay linearSystem linearSolverLapack linearSolverTotalPivot mixedSystem mixedSearchSolver nonlinearSystem nonlinearValuesList nonlinearSolverHybrd nonlinearSolverHomotopy omc_math model_help stateset synchronous)
            list(APPEND units "${runtime}/simulation/solver/${name}.c")
        endforeach()
    else()
        message(FATAL_ERROR "Unknown recipe ${recipe}")
    endif()

    add_library(${target} SHARED ${units})
    target_include_directories(${target} PRIVATE ${includes})
    target_compile_definitions(${target} PRIVATE ${definitions})
    if(NOT WIN32)
        target_link_libraries(${target} PRIVATE m)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(${target} PRIVATE -Wno-deprecated-declarations)
    endif()
    if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_link_options(${target} PRIVATE -static -static-libgcc -static-libstdc++)
        if(recipe STREQUAL "openmodelica")
            target_link_libraries(${target} PRIVATE -Wl,-Bstatic winpthread -Wl,-Bdynamic)
        endif()
    endif()
    set_target_properties(${target} PROPERTIES
        PREFIX "" OUTPUT_NAME "${identifier}"
        ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/${target}"
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/${target}"
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/${target}")
    if(APPLE AND path MATCHES "^fmi2/" AND FMU_MACOS_UNIVERSAL_FMI2)
        set_target_properties(${target} PROPERTIES OSX_ARCHITECTURES "x86_64;arm64")
        if(BUILD_TESTING)
            foreach(arch x86_64 arm64)
                add_test(NAME ${target}_${arch} COMMAND lipo "$<TARGET_FILE:${target}>" -verify_arch ${arch})
            endforeach()
        endif()
    endif()
    if(APPLE AND path MATCHES "^fmi3/")
        if(fmi3_arch STREQUAL "aarch64")
            set_target_properties(${target} PROPERTIES OSX_ARCHITECTURES arm64)
        else()
            set_target_properties(${target} PROPERTIES OSX_ARCHITECTURES x86_64)
        endif()
    endif()
    add_custom_target(${target}_package ALL
        COMMAND "${CMAKE_COMMAND}" "-DMODEL=${model}" "-DBINARY=$<TARGET_FILE:${target}>"
            "-DPLATFORM=${platform}" "-DOUTPUT=${CMAKE_BINARY_DIR}/fmus/${path}.fmu"
            "-DSTAGE=${CMAKE_BINARY_DIR}/stage/${path}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/PackageFmu.cmake"
        DEPENDS ${target} VERBATIM)
    if(BUILD_TESTING)
        if(path MATCHES "^fmi2/")
            set(version 2)
        else()
            set(version 3)
        endif()
        add_test(NAME ${target}_load COMMAND fmu_smoke "$<TARGET_FILE:${target}>" ${version})
        add_test(NAME ${target}_archive COMMAND "${CMAKE_COMMAND}"
            "-DARCHIVE=${CMAKE_BINARY_DIR}/fmus/${path}.fmu" "-DPLATFORM=${platform}"
            "-DBINARY_NAME=$<TARGET_FILE_NAME:${target}>" "-DVERIFY_DIR=${CMAKE_BINARY_DIR}/verify/${path}"
            -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/VerifyFmu.cmake")
    endif()
endfunction()
